```python
import minescript as m
import time

"""
🎣 Fishing Debug Tool for Minescript 4.0 / JE 1.20.1
サーバー側釣り挙動を調査するためのデバッグスクリプト
"""

def find_bobbers():
    """全エンティティのうち、浮きっぽいものをリストで返す"""
    bobbers = []
    for e in m.entities():
        name = e.type.lower()
        if "bobber" in name or "hook" in name or "fishing" in name:
            bobbers.append(e)
    return bobbers

def debug_loop():
    m.echo("=== 🎣 Fishing Debugger Started ===")
    m.echo("1. 釣り竿を投げてみてください")
    m.echo("2. 約10〜30秒間ログを観察します（他プレイヤーには表示されません）")

    last_bobbers = {}
    tick = 0
    while True:
        tick += 1
        bobbers = find_bobbers()

        # --- 1. 浮きの数を確認 ---
        if len(bobbers) == 0:
            if tick % 10 == 0:
                m.echo("[No bobber detected]")
        else:
            for b in bobbers:
                y = b.position[1]
                t = b.type

                # 前回位置との差分
                diff = 0
                if b.id in last_bobbers:
                    diff = y - last_bobbers[b.id]
                last_bobbers[b.id] = y

                # --- 2. ログ出力 ---
                if tick % 5 == 0:
                    m.echo(f"[{t}] Y={y:.3f} Δ={diff:+.3f}")

        # --- 3. 消えた浮きの検出 ---
        disappeared = [bid for bid in list(last_bobbers.keys()) if bid not in [b.id for b in bobbers]]
        for bid in disappeared:
            m.echo(f"🪝 Bobber {bid} disappeared (可能: 魚がかかった or 回収された)")
            del last_bobbers[bid]

        time.sleep(0.1)

if __name__ == "__main__":
    debug_loop()
```

```python
import minescript as m
import time

"""
🎣 Fishing Debugger (Auto Cast + Manual Reel)
- 実行直後に自動で釣竿を投げる
- 浮きIDを自動検出して追跡
- 手動で釣り上げてOK（浮きが消えるまで監視）
- JE 1.20.1 / Minescript 4.0
"""

CHECK_INTERVAL = 0.05
SINK_THRESHOLD = -0.3
STABILIZE_THRESHOLD = 0.02
STABILIZE_COUNT = 5


def find_newest_bobber():
    bobbers = [e for e in m.entities() if "fishing_bobber" in e.type.lower()]
    return max(bobbers, key=lambda e: e.id) if bobbers else None


def cast_fishing_rod():
    """釣竿を1回投げる"""
    m.player_press_use(True)
    time.sleep(0.15)
    m.player_press_use(False)
    m.echo("🎯 釣り竿を投げました。浮きを検出中...")


def main():
    m.echo("=== 🎣 Fishing Debugger (Auto Cast + Manual Reel) ===")
    m.echo("実行後に自動で釣り竿を投げます。")
    m.echo("自分で引き上げてOKです。浮きの動きを追跡します。")

    time.sleep(1.0)  # 1秒待機して環境安定
    cast_fishing_rod()

    tracked_id = None
    last_y = None
    stable_counter = 0

    # --- 浮きが出るまで待機 ---
    start_time = time.time()
    while tracked_id is None and time.time() - start_time < 5:
        newest = find_newest_bobber()
        if newest:
            tracked_id = newest.id
            last_y = newest.position[1]
            m.echo(f"✅ 浮きを検出: ID={tracked_id}")
        time.sleep(0.1)

    if not tracked_id:
        m.echo("⚠️ 浮きが見つかりません。水辺に向いて投げ直してください。")
        return

    # --- メイン監視ループ ---
    while True:
        bobbers = [e for e in m.entities() if "fishing_bobber" in e.type.lower()]
        bobber = next((b for b in bobbers if b.id == tracked_id), None)

        # 浮きが消えた → 手動リール or 魚ヒット
        if not bobber:
            m.echo(f"💨 Bobber {tracked_id} disappeared (manual reel or bite)")
            break

        y = bobber.position[1]
        diff = y - last_y

        # 浮きの動き解析
        if abs(diff) > 0.001:
            if diff <= SINK_THRESHOLD:
                m.echo(f"⬇️ Bobber {tracked_id} sinking (ΔY={diff:.3f})")
                stable_counter = 0
            elif abs(diff) < STABILIZE_THRESHOLD:
                stable_counter += 1
                if stable_counter == STABILIZE_COUNT:
                    m.echo(f"🟦 Bobber {tracked_id} stabilized (ΔY={diff:.3f})")
            else:
                stable_counter = 0

        last_y = y
        time.sleep(CHECK_INTERVAL)

    m.echo("✅ デバッグ終了。")


if __name__ == "__main__":
    main()
```

```python
import minescript as m
import time

"""
🎣 Fishing Debugger (Endless Auto Cast + Manual Reel OK)
- 自動で釣竿を投げる → 浮きの動きを追跡 → 消えたら再キャスト
- 手動で引き上げてもOK（次キャストへ移行）
- JE 1.20.1 / Minescript 4.0
"""

CHECK_INTERVAL = 0.05          # 監視間隔（秒）
SINK_THRESHOLD = -0.3          # 沈み判定しきい値
STABILIZE_THRESHOLD = 0.02     # 安定判定しきい値
STABILIZE_COUNT = 5            # 連続安定回数で安定判定
CAST_INTERVAL = 1.0            # 浮き消失後の再キャストまでの待機秒数


def find_newest_bobber():
    """最新の浮きエンティティを取得"""
    bobbers = [e for e in m.entities() if "fishing_bobber" in e.type.lower()]
    return max(bobbers, key=lambda e: e.id) if bobbers else None


def cast_fishing_rod():
    """釣竿を1回投げる"""
    m.player_press_use(True)
    time.sleep(0.15)
    m.player_press_use(False)
    m.echo("🎯 釣り竿を投げました。浮きを検出中...")


def monitor_bobber(tracked_id: int):
    """浮きの動きを追跡。消えたら終了。"""
    last_y = None
    stable_counter = 0
    tick = 0

    while True:
        bobbers = [e for e in m.entities() if "fishing_bobber" in e.type.lower()]
        bobber = next((b for b in bobbers if b.id == tracked_id), None)

        if not bobber:
            m.echo(f"💨 浮き {tracked_id} が消えました（リール or 魚ヒット）")
            return  # 消失＝次のキャストへ

        y = bobber.position[1]
        if last_y is not None:
            diff = y - last_y
            tick += 1

            # 浮きの挙動をログ出力
            if abs(diff) > 0.001:
                if diff <= SINK_THRESHOLD:
                    m.echo(f"[{tick:04d}] ⬇️ 浮き沈み ΔY={diff:.3f}")
                    stable_counter = 0
                elif abs(diff) < STABILIZE_THRESHOLD:
                    stable_counter += 1
                    if stable_counter == STABILIZE_COUNT:
                        m.echo(f"[{tick:04d}] 🟦 安定 ΔY={diff:.3f}")
                else:
                    stable_counter = 0

        last_y = y
        time.sleep(CHECK_INTERVAL)


def main():
    m.echo("=== 🎣 Fishing Debugger (Endless Auto Cast) ===")
    m.echo("💡 自動で投げ続け、浮きの動きを常時ログします。")
    m.echo("💡 手動でリールしてもOK。")

    loop_count = 0
    time.sleep(1.0)

    while True:
        loop_count += 1
        m.echo(f"\n=== 🎣 第 {loop_count} 投目 開始 ===")

        cast_fishing_rod()

        # 浮きが出現するまで待つ
        tracked_id = None
        start_time = time.time()
        while tracked_id is None and time.time() - start_time < 5:
            newest = find_newest_bobber()
            if newest:
                tracked_id = newest.id
                m.echo(f"✅ 浮きを検出: ID={tracked_id}")
            time.sleep(0.1)

        if not tracked_id:
            m.echo("⚠️ 浮きが見つかりません。少し待って再試行します。")
            time.sleep(2.0)
            continue

        # 浮きを監視
        monitor_bobber(tracked_id)

        # 消失後 → 再キャストへ
        m.echo(f"⏳ {CAST_INTERVAL:.1f}秒後に次のキャストを開始...")
        time.sleep(CAST_INTERVAL)


if __name__ == "__main__":
    main()
```


### chatlog

```python
import minescript as m
from minescript import EventQueue, EventType
import datetime

LOG_FILE = "chat_log.txt"

def log_message(msg: str):
    """Append chat messages to a log file"""
    timestamp = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    with open(LOG_FILE, "a", encoding="utf-8") as f:
        f.write(f"[{timestamp}] {msg}\n")

eq = EventQueue()
eq.register_chat_listener()
#m.echo("Chat listener ready! Logging ALL chat messages...")

while True:
    event = eq.get()
    if not event or event.type != EventType.CHAT:
        continue

    msg = event.message
    log_message(msg)
```
